CREATE TABLE SANDBOX_DB.SANDBOX_SCHEMA.MEMBER_MONTHLY_POP AS

WITH DEMO_DATES AS (

    SELECT DISTINCT POP_DATE
    FROM CUSTOMER_DEMO
    WHERE POP_DATE BETWEEN '2025-10-01' AND '2026-10-31'

),

FEATURE_DATES AS (

    SELECT DISTINCT POP_DATE
    FROM MEMBER_FEATURES
    WHERE POP_DATE BETWEEN '2025-10-01' AND '2026-10-31'

),

MONTHLY_DATES AS (

    SELECT
        LAST_DAY(F.POP_DATE) AS MONTH_END,
        D.POP_DATE AS DEMO_DATE,
        F.POP_DATE AS FEATURE_DATE

    FROM DEMO_DATES D

    INNER JOIN FEATURE_DATES F
        ON F.POP_DATE = DATEADD(DAY, 1, D.POP_DATE)

    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY LAST_DAY(F.POP_DATE)
        ORDER BY F.POP_DATE DESC
    ) = 1

),

BASE_POP AS (

    SELECT
        M.MONTH_END AS POP_DATE,
        D.MEMBER_ID,
        D.ACTIVE_IND,
        F.CURRENT_BANKPROD_CNT,
        M.DEMO_DATE,
        M.FEATURE_DATE

    FROM MONTHLY_DATES M

    INNER JOIN CUSTOMER_DEMO D
        ON D.POP_DATE = M.DEMO_DATE

    INNER JOIN MEMBER_FEATURES F
        ON F.MEMBER_ID = D.MEMBER_ID
        AND F.POP_DATE = M.FEATURE_DATE

    WHERE D.ACTIVE_IND = 1
        AND F.CURRENT_BANKPROD_CNT >= 1
        AND D.MEMBER_ID IS NOT NULL

),

MONTHLY_TASKS AS (

    SELECT
        MEMBER_ID,
        LAST_DAY(TASK_DATE) AS TASK_MONTH

    FROM TASK_ACTIVITY

    WHERE TASK_DATE >= '2024-11-01'
        AND TASK_DATE < '2026-11-01'
        AND MEMBER_ID IS NOT NULL

    GROUP BY 1, 2

),

FINAL_POP AS (

    SELECT
        B.*,
        T.TASK_MONTH AS LAST_TASK_MONTH,

        CASE
            WHEN T.TASK_MONTH >= DATEADD(
                MONTH, -11, DATE_TRUNC('MONTH', B.POP_DATE)
            )
            THEN 1
            ELSE 0
        END AS TASK_FLAG_12M

    FROM BASE_POP B

    ASOF JOIN MONTHLY_TASKS T
        MATCH_CONDITION(B.POP_DATE >= T.TASK_MONTH)
        ON B.MEMBER_ID = T.MEMBER_ID

)

SELECT
    *,
    CASE
        WHEN TASK_FLAG_12M = 1 THEN 'POP 1'
        ELSE 'POP 2'
    END AS TASK_POPULATION

FROM FINAL_POP;

SELECT
    A.POP_DATE AS PREVIOUS_MONTH,
    B.POP_DATE AS CURRENT_MONTH,

    COUNT_IF(A.TASK_FLAG_12M = 1
         AND B.TASK_FLAG_12M = 1) AS POP_1_TO_1,

    COUNT_IF(A.TASK_FLAG_12M = 1
         AND B.TASK_FLAG_12M = 0) AS POP_1_TO_2,

    COUNT_IF(A.TASK_FLAG_12M = 0
         AND B.TASK_FLAG_12M = 1) AS POP_2_TO_1,

    COUNT_IF(A.TASK_FLAG_12M = 0
         AND B.TASK_FLAG_12M = 0) AS POP_2_TO_2,

    ROUND(100.0 * POP_1_TO_2 /
        NULLIF(POP_1_TO_1 + POP_1_TO_2, 0), 2)
        AS INACTIVATION_PCT,

    ROUND(100.0 * POP_2_TO_1 /
        NULLIF(POP_2_TO_1 + POP_2_TO_2, 0), 2)
        AS REACTIVATION_PCT

FROM MEMBER_MONTHLY_POP A

INNER JOIN MEMBER_MONTHLY_POP B
    ON A.MEMBER_ID = B.MEMBER_ID
    AND B.POP_DATE = LAST_DAY(
        DATEADD(MONTH, 1, A.POP_DATE)
    )

GROUP BY 1, 2
ORDER BY 1, 2;

Tab 5 — Base Population Changes
WITH START_POP AS (

    SELECT MEMBER_ID
    FROM MEMBER_MONTHLY_POP
    WHERE POP_DATE = '2025-10-31'

),

END_POP AS (

    SELECT MEMBER_ID
    FROM MEMBER_MONTHLY_POP
    WHERE POP_DATE = '2026-10-31'

)

SELECT
    CASE
        WHEN A.MEMBER_ID IS NULL THEN 'Newly Eligible'
        WHEN B.MEMBER_ID IS NULL THEN 'No Longer Eligible'
        ELSE 'Eligible at Both Dates'
    END AS MEMBER_STATUS,

    COUNT(*) AS MEMBER_COUNT

FROM START_POP A

FULL OUTER JOIN END_POP B
    ON A.MEMBER_ID = B.MEMBER_ID

GROUP BY 1
ORDER BY 1;

Tab 6 — Population Stability
WITH MEMBER_SUMMARY AS (

    SELECT
        MEMBER_ID,
        COUNT(DISTINCT POP_DATE) AS MONTHS_PRESENT,
        SUM(TASK_FLAG_12M) AS ACTIVE_MONTHS
    FROM MEMBER_MONTHLY_POP
    GROUP BY 1

)

SELECT
    CASE
        WHEN ACTIVE_MONTHS = 13 THEN 'Always Pop 1'
        WHEN ACTIVE_MONTHS = 0 THEN 'Always Pop 2'
        ELSE 'Changed Population'
    END AS POPULATION_STATUS,

    COUNT(*) AS MEMBER_COUNT,

    ROUND(
        100.0 * COUNT(*) / SUM(COUNT(*)) OVER (),
        2
    ) AS MEMBER_PCT

FROM MEMBER_SUMMARY

WHERE MONTHS_PRESENT = 13

GROUP BY 1
ORDER BY 1;

One Excel formatting note: The SQL percentage columns already return values such as 23.00, not 0.23. Keep them as numbers with a percentage column heading, or use Excel's custom format 0.00"%". Otherwise, standard percentage formatting will display 2300%.
